cmake_minimum_required(VERSION 3.20)
project(tibber_pulse_host_tests CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_compile_options(-Wall -Wextra -Wpedantic -Werror)

include(FetchContent)
FetchContent_Declare(
  googletest
  URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.tar.gz
  URL_HASH SHA256=8ad598c73ad796e0d8280b082cebd82a630d73e73cd3c70057938a6501bba5d7
)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

# Repo root, two levels up from tests/components/tibber_pulse.
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)

# The SML decoder, against telegrams the Python decoder was run on
# (host_sml_vectors.h, generated by _gen_sml_vectors.py). Built with
# AddressSanitizer: it parses bytes off the network, so an out-of-bounds read
# on a corrupt telegram is the bug worth catching, and a plain build would not.
add_executable(host_sml_power_test
  host_sml_power_test.cpp
  ${REPO_ROOT}/esphome/components/tibber_pulse/sml_power.cpp
)
target_include_directories(host_sml_power_test PRIVATE
  ${REPO_ROOT}
  ${CMAKE_CURRENT_SOURCE_DIR}
)
target_compile_options(host_sml_power_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(host_sml_power_test PRIVATE -fsanitize=address,undefined)
target_link_libraries(host_sml_power_test PRIVATE GTest::gtest_main)

# The /node_data.json <-> /data.json fallback. No ESPHome deps.
add_executable(host_bridge_endpoint_test
  host_bridge_endpoint_test.cpp
  ${REPO_ROOT}/esphome/components/tibber_pulse/bridge_endpoint.cpp
)
target_include_directories(host_bridge_endpoint_test PRIVATE
  ${REPO_ROOT}
  ${CMAKE_CURRENT_SOURCE_DIR}
)
target_link_libraries(host_bridge_endpoint_test PRIVATE GTest::gtest_main)

enable_testing()
include(GoogleTest)
gtest_discover_tests(host_sml_power_test)
gtest_discover_tests(host_bridge_endpoint_test)
